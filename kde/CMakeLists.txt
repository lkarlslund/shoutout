cmake_minimum_required(VERSION 3.20)
project(shoutout_settings LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
add_library(shoutout_settings_panel STATIC settingspanel.h)
set_target_properties(shoutout_settings_panel PROPERTIES LINKER_LANGUAGE CXX)
target_link_libraries(shoutout_settings_panel PUBLIC Qt6::Core Qt6::Widgets Qt6::Network)
add_executable(shoutout-settings main.cpp)
set_target_properties(shoutout-settings PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/..")
target_link_libraries(shoutout-settings PRIVATE shoutout_settings_panel)
install(TARGETS shoutout-settings DESTINATION bin COMPONENT core)
add_executable(settings_smoke smoke.cpp)
target_link_libraries(settings_smoke PRIVATE shoutout_settings_panel)

option(SHOUTOUT_KDE "Build the optional KDE System Settings integration" ON)
if(SHOUTOUT_KDE)
  find_package(KF6KCMUtils QUIET)
  find_package(KF6CoreAddons QUIET)
  if(KF6KCMUtils_FOUND AND KF6CoreAddons_FOUND)
    add_library(kcm_shoutout MODULE kcm.cpp)
    set_target_properties(kcm_shoutout PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/..")
    target_link_libraries(kcm_shoutout PRIVATE shoutout_settings_panel KF6::KCMUtils KF6::CoreAddons)
    install(TARGETS kcm_shoutout DESTINATION lib/qt6/plugins/plasma/kcms/systemsettings_qwidgets COMPONENT kde)
    add_executable(kde_smoke smoke.cpp)
    target_compile_definitions(kde_smoke PRIVATE SHOUTOUT_KDE_SMOKE)
    target_link_libraries(kde_smoke PRIVATE Qt6::Core Qt6::Widgets Qt6::Network KF6::KCMUtils KF6::CoreAddons)
  else()
    message(STATUS "KDE frameworks not found: building standalone settings only")
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(settings_test settings_test.cpp)
  target_link_libraries(settings_test PRIVATE shoutout_settings_panel)
  add_test(NAME settings COMMAND settings_test)
  set_tests_properties(settings PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
